    <script>
        // مفاتيح الـ API الخاصة بك
        const TMDB_API_KEY = 'e7be99b2666a862f16f0a6b5441c150b'; 
        const OMDB_API_KEY = '75583472';

        const BASE_URL = 'https://api.themoviedb.org/3';
        const IMG_URL = 'https://image.tmdb.org/t/p/w500';
        
        const grid = document.getElementById('shows-grid');
        const modal = document.getElementById('show-modal');

        let currentNetwork = '213'; // لحفظ المنصة المحددة حالياً

        // دالة تطبيق الترتيب عند تغيير القائمة المنسدلة
        function applySort() {
            const activeBtn = document.querySelector('.filters button.active');
            if (activeBtn) {
                // إذا كنا نستعرض منصة، نعيد جلب البيانات بالترتيب الجديد
                fetchShows(currentNetwork, activeBtn);
            } else if (document.getElementById('search-input').value.trim() !== '') {
                // إذا كنا في وضع البحث، نعيد ترتيب نتائج البحث
                performSearch();
            }
        }

        // Fetch Shows by Network
        async function fetchShows(networkId, btnElement) {
            currentNetwork = networkId; // حفظ المنصة الحالية
            document.getElementById('search-input').value = ''; 
            
            if (btnElement) {
                const buttons = document.querySelectorAll('.filters button');
                buttons.forEach(btn => btn.classList.remove('active'));
                btnElement.classList.add('active');
            }

            const sortValue = document.getElementById('sort-select').value;
            
            // إضافة فلتر إضافي لمنع ظهور مسلسلات مستقبلية فارغة أو بدون تقييم عند اختيار الترتيب بالتاريخ أو التقييم
            let extraParams = '';
            if (sortValue === 'vote_average.desc') {
                extraParams = '&vote_count.gte=200'; // يفضل أن يكون التقييم مبني على 200 صوت على الأقل
            } else if (sortValue === 'first_air_date.desc') {
                extraParams = '&vote_count.gte=5'; // لمنع المسلسلات غير المصدرة بعد والتي ليس لها صور
            }

            grid.innerHTML = '<div class="loading">Loading shows... ⏳</div>';

            try {
                const response = await fetch(`${BASE_URL}/discover/tv?api_key=${TMDB_API_KEY}&language=en-US&sort_by=${sortValue}${extraParams}&with_networks=${networkId}`);
                if (!response.ok) throw new Error('Connection error with TMDB.');

                const data = await response.json();
                displayShows(data.results);
            } catch (error) {
                grid.innerHTML = `<div class="loading" style="color: red;">${error.message}</div>`;
            }
        }

        // Search Functionality
        async function performSearch() {
            const query = document.getElementById('search-input').value.trim();
            if (!query) return;

            const buttons = document.querySelectorAll('.filters button');
            buttons.forEach(btn => btn.classList.remove('active'));

            grid.innerHTML = '<div class="loading">Searching... 🔍</div>';

            try {
                const response = await fetch(`${BASE_URL}/search/tv?api_key=${TMDB_API_KEY}&language=en-US&query=${encodeURIComponent(query)}`);
                if (!response.ok) throw new Error('Search failed.');

                const data = await response.json();
                
                if (data.results.length === 0) {
                    grid.innerHTML = '<div class="loading">No shows found. Try another name!</div>';
                    return;
                }

                // ترتيب نتائج البحث يدوياً (Client-side sorting)
                const sortValue = document.getElementById('sort-select').value;
                let sortedResults = data.results;
                
                if (sortValue === 'first_air_date.desc') {
                    // ترتيب حسب الأحدث
                    sortedResults.sort((a, b) => new Date(b.first_air_date || 0) - new Date(a.first_air_date || 0));
                } else if (sortValue === 'vote_average.desc') {
                    // ترتيب حسب التقييم
                    sortedResults.sort((a, b) => b.vote_average - a.vote_average);
                } else {
                    // ترتيب حسب الشهرة (الافتراضي)
                    sortedResults.sort((a, b) => b.popularity - a.popularity);
                }

                displayShows(sortedResults);
            } catch (error) {
                grid.innerHTML = `<div class="loading" style="color: red;">${error.message}</div>`;
            }
        }

        function handleKeyPress(event) {
            if (event.key === 'Enter') {
                performSearch();
            }
        }

        function displayShows(showsToDisplay) {
            grid.innerHTML = '';
            
            showsToDisplay.forEach(show => {
                const card = document.createElement('div');
                card.className = 'show-card';
                card.onclick = () => openModal(show);
                
                const posterPath = show.poster_path ? IMG_URL + show.poster_path : 'https://via.placeholder.com/500x750?text=No+Image';
                
                // إضافة تاريخ الإصدار للبطاقة ليراه المستخدم عند الترتيب بالتاريخ
                const releaseYear = show.first_air_date ? show.first_air_date.substring(0,4) : 'N/A';

                card.innerHTML = `
                    <img src="${posterPath}" alt="${show.name}" class="show-image">
                    <div class="show-info">
                        <h3 class="show-title">${show.name}</h3>
                        <p style="color: #aaa; font-size: 0.85rem; margin: 0; display: flex; justify-content: space-between;">
                            <span>⭐ ${show.vote_average.toFixed(1)}</span>
                            <span>📅 ${releaseYear}</span>
                        </p>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        // Fetch IMDb Rating using OMDb API
        async function fetchIMDbRating(tmdbId) {
            try {
                const extRes = await fetch(`${BASE_URL}/tv/${tmdbId}/external_ids?api_key=${TMDB_API_KEY}`);
                const extData = await extRes.json();
                const imdbId = extData.imdb_id;

                if (!imdbId) return "No IMDb ID available";

                const omdbRes = await fetch(`https://www.omdbapi.com/?i=${imdbId}&apikey=${OMDB_API_KEY}`);
                const omdbData = await omdbRes.json();

                if (omdbData.Response === "True" && omdbData.imdbRating !== "N/A") {
                    return `${omdbData.imdbRating} / 10 (${omdbData.imdbVotes} votes)`;
                }
                return "Not available";
            } catch (error) {
                return "Error loading IMDb data";
            }
        }

        // --- Modal Functions ---
        async function openModal(show) {
            document.getElementById('modal-title').innerText = show.name;
            document.getElementById('modal-rating').innerText = show.vote_average.toFixed(1);
            document.getElementById('modal-date').innerText = 'Release Date: ' + (show.first_air_date || 'Not available');
            document.getElementById('modal-overview').innerText = show.overview ? show.overview : 'Sorry, no overview available for this show.';
            
            const posterPath = show.poster_path ? IMG_URL + show.poster_path : 'https://via.placeholder.com/500x750?text=No+Image';
            document.getElementById('modal-img').src = posterPath;
            
            const imdbRatingSpan = document.getElementById('modal-imdb-rating');
            imdbRatingSpan.innerText = 'Loading... ⏳';
            imdbRatingSpan.style.color = '#aaa';

            modal.style.display = 'flex';

            const imdbRating = await fetchIMDbRating(show.id);
            imdbRatingSpan.innerText = imdbRating;
            imdbRatingSpan.style.color = 'white';
        }

        function closeModal() {
            modal.style.display = 'none';
        }

        window.onclick = function(event) {
            if (event.target == modal) {
                closeModal();
            }
        }

        // Initial fetch
        fetchShows('213', document.querySelector('.filters button'));
    </script>
